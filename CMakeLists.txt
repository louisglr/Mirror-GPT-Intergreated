cmake_minimum_required(VERSION 3.22)

project(MirrorPlugin VERSION 1.0.0)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)
FetchContent_Declare(
    JUCE
    GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    # JUCE 9 supports the current macOS/Xcode SDK used by GitHub Actions.
    GIT_TAG        9.0.1
    GIT_SHALLOW    TRUE
)
FetchContent_MakeAvailable(JUCE)

juce_add_plugin(Mirror
    COMPANY_NAME "Louis Gabriel"
    PLUGIN_MANUFACTURER_CODE LoGa
    PLUGIN_CODE Mirr
    FORMATS AU VST3
    PRODUCT_NAME "MIRROR"
    BUNDLE_ID "com.louisgabriel.mirror"
    IS_SYNTH FALSE
    NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE
)

juce_add_binary_data(MirrorAssets
    SOURCES
        Assets/mirror_engraved_background.jpg
        Assets/mirror_reference_main.png
        Assets/mirror_reference_harmony.png
        Assets/mirror_reference_advanced.png
)

target_sources(Mirror PRIVATE
    Source/PluginProcessor.cpp
    Source/PluginEditor.cpp
)

target_compile_definitions(Mirror PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
)

target_link_libraries(Mirror PRIVATE
    juce::juce_audio_utils
    juce::juce_dsp
    MirrorAssets
    juce::juce_recommended_config_flags
    juce::juce_recommended_lto_flags
    juce::juce_recommended_warning_flags
)
